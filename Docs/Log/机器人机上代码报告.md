## 9. 机器人配置 (configuration.toml)

路径: `/home/ubuntu/Documents/configuration.toml`

### 关键配置

```toml
[basic]
station_type = "TienYi2_ROS2"
control_rate_hz = 100
tele_agent_type = "ServoTeleCore"

[hand.left]
enable = true
type = "InspireHand3Ros2"
control_mode = "manus"

[hand.right]
enable = true
type = "InspireHand3Ros2"
control_mode = "manus"

[robot.arm.home]
robot = [0.0, 0.08, 0.0, -0.16, 0.01, 0.0, 0.0,
         0.0, -0.08, 0.0, -0.16, -0.01, 0.0, 0.0]
[robot.head]
home = [0.0, 0.4, 0.0]
[robot.controller]
enable_head_controller = true
enable_dual_arm_controller = true
enable_waist_controller = true
enable_leg_controller = true
sub_float_base_rpyz_status = true
enable_hybrid_controller = true
enable_ros2control = false  # 不使用 ros2_control

[force_sensor.left]
enable = false  # 注意: 实际使用中通过 bodyctrl 直接读取力传感器
[force_sensor.right]
enable = false

[chassis]
enable = false  # 底盘不启用

[camera.camera]
enable = true
type = "OrbbecCameraRos2"
camera_model = "orbbec"
camera_color_resolution = [640, 480]
camera_depth_resolution = [640, 480]

[data_collect]
save_type = "hdf5"
save_path = "~/Data/sim_data"
robot_state_save_rate_hz = 60
camera_state_save_rate_hz = 30
```
